cmake_minimum_required(VERSION 3.20)
project(SensorUdp VERSION 1.0 LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This project targets Windows and Winsock2")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/W4 /permissive- /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
add_library(sensor_core src/protocol.cpp src/sensor_state.cpp src/sequence_tracker.cpp)
target_include_directories(sensor_core PUBLIC include)
add_library(sensor_platform src/udp_socket.cpp src/options.cpp src/console_input.cpp)
target_include_directories(sensor_platform PUBLIC include)
target_compile_definitions(sensor_platform PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601)
target_link_libraries(sensor_platform PUBLIC sensor_core ws2_32)
add_executable(SensorSim src/sensor_main.cpp)
target_link_libraries(SensorSim PRIVATE sensor_platform)
add_executable(Collector src/collector_main.cpp)
target_link_libraries(Collector PRIVATE sensor_platform)
include(CTest)
if(BUILD_TESTING)
    add_executable(unit_tests tests/unit_tests.cpp)
    target_link_libraries(unit_tests PRIVATE sensor_core)
    add_test(NAME unit COMMAND unit_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME integration COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py"
            --bin-dir "$<TARGET_FILE_DIR:SensorSim>"
            --output-dir "${CMAKE_CURRENT_BINARY_DIR}/test-results")
        set_tests_properties(integration PROPERTIES TIMEOUT 60)
    endif()
endif()
